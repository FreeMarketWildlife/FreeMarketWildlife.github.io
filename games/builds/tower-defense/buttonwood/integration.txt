function bwIndexTerrain(){

  if(!z.game?.placed||z.game.naturalTree||!artPaletteV32[z.game.material]||(!z.isStatic&&!z.isSleeping))continue;
  const cols=Math.max(1,Math.round(z.game.w/B)),rows=Math.max(1,Math.round(z.game.h/B));
  const ox=z.position.x-cols*B/2,oy=z.position.y-rows*B/2,cx=Math.round(ox/B),cy=Math.round(oy/B);
  const angle=Math.atan2(Math.sin(z.angle),Math.cos(z.angle));

  bwAlignedBodies.set(z,{cx,cy});
  for(let y=0;y<rows;y++)for(let x=0;x<cols;x++){
   const k=key(cx+x,cy+y),existing=bwPlacedCells.get(k);
