let fogFieldV72=null,fogSeedsV72=null;
function fogOpenV72(cx,cy){

 const k=key(cx,cy);if(grid.has(k))return false;
 if(removedTerrain.has(k))return true;

}
function fogPrepareV72(){

function fogLayerV72(cx,cy){return cy<0?0:fogFieldV72?.at(cx,cy)??fogArtV72.maxLayer}
